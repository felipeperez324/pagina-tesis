-- Estructura:
--   Usuarios
--   Mascotas
--   Sesiones
--   Verificación de correo
--   Postulaciones
--   Preferencias de Match
--

BEGIN;


-- ============================================================
-- USUARIOS
-- ============================================================

CREATE TABLE IF NOT EXISTS public.usuarios (

    id SERIAL PRIMARY KEY,

    nombre VARCHAR(100) NOT NULL,

    rut VARCHAR(12) NOT NULL UNIQUE,

    correo VARCHAR(150) NOT NULL UNIQUE,

    tipo_usuario VARCHAR(20) NOT NULL
        CHECK (
            tipo_usuario IN (
                'adoptante',
                'organizacion'
            )
        ),

    contrasena VARCHAR(255) NOT NULL,

    estado_cuenta VARCHAR(20) NOT NULL
        DEFAULT 'activo'
        CHECK (
            estado_cuenta IN (
                'activo',
                'suspendido'
            )
        ),

    correo_verificado BOOLEAN NOT NULL
        DEFAULT FALSE,

    descripcion_perfil TEXT NOT NULL
        DEFAULT '',

    fecha_registro TIMESTAMPTZ NOT NULL
        DEFAULT NOW(),

    correo_pendiente VARCHAR(150),

    correo_codigo_hash TEXT,

    correo_codigo_expira TIMESTAMPTZ,

    correo_codigo_enviado TIMESTAMPTZ,

    correo_codigo_intentos INTEGER NOT NULL
        DEFAULT 0

);


-- ============================================================
-- MASCOTAS
-- ============================================================

CREATE TABLE IF NOT EXISTS public.mascotas (

    id SERIAL PRIMARY KEY,

    usuario_id INTEGER NOT NULL
        REFERENCES public.usuarios(id)
        ON DELETE CASCADE,

    nombre VARCHAR(100) NOT NULL,

    descripcion TEXT,

    personalidad TEXT,

    personalidad_codigo VARCHAR(40)
        CHECK (
            personalidad_codigo IS NULL
            OR personalidad_codigo IN (
                'tranquilo',
                'carinoso',
                'jugueton',
                'activo',
                'timido',
                'sociable',
                'independiente',
                'protector',

                'tranquilo_carinoso',
                'activo_jugueton',
                'jugueton_sociable',
                'timido_carinoso',
                'independiente_tranquilo',

                'otro'
            )
        ),

    historial_medico TEXT,

    salud_codigo VARCHAR(40)
        CHECK (
            salud_codigo IS NULL
            OR salud_codigo IN (
                'sin_antecedentes',
                'vacunacion_al_dia',
                'vacunacion_pendiente',
                'tratamiento',
                'condicion_cronica',
                'cuidados_especiales',
                'desconocido',
                'otro'
            )
        ),

    estado_adopcion VARCHAR(20) NOT NULL
        DEFAULT 'disponible'
        CHECK (
            estado_adopcion IN (
                'disponible',
                'reservada',
                'adoptada'
            )
        ),

    tipo_mascota VARCHAR(20) NOT NULL
        CHECK (
            tipo_mascota IN (
                'perro',
                'gato',
                'otro'
            )
        )

);


-- ============================================================
-- SESIONES
-- ============================================================

CREATE TABLE IF NOT EXISTS public.sesiones (

    id UUID PRIMARY KEY,

    usuario_id INTEGER NOT NULL
        REFERENCES public.usuarios(id)
        ON DELETE CASCADE,

    fecha_creacion TIMESTAMPTZ NOT NULL
        DEFAULT NOW(),

    ultima_actividad TIMESTAMPTZ NOT NULL
        DEFAULT NOW()

);


-- ============================================================
-- CÓDIGOS DE VERIFICACIÓN DE USUARIOS
-- ============================================================

CREATE TABLE IF NOT EXISTS public.codigos_verificacion (

    usuario_id INTEGER PRIMARY KEY
        REFERENCES public.usuarios(id)
        ON DELETE CASCADE,

    codigo_hash VARCHAR(64) NOT NULL,

    expira_en TIMESTAMPTZ NOT NULL,

    intentos INTEGER NOT NULL DEFAULT 0,

    ultimo_envio TIMESTAMPTZ NOT NULL
        DEFAULT NOW(),

    ventana_iniciada TIMESTAMPTZ NOT NULL
        DEFAULT NOW(),

    envios_ventana INTEGER NOT NULL DEFAULT 1

);


-- ============================================================
-- VERIFICACIONES DE REGISTRO
-- ============================================================

CREATE TABLE IF NOT EXISTS public.verificaciones_registro (

    correo VARCHAR(150) PRIMARY KEY,

    codigo_hash VARCHAR(64) NOT NULL,

    expira_en TIMESTAMPTZ NOT NULL,

    intentos INTEGER NOT NULL DEFAULT 0,

    ultimo_envio TIMESTAMPTZ NOT NULL
        DEFAULT NOW(),

    ventana_iniciada TIMESTAMPTZ NOT NULL
        DEFAULT NOW(),

    envios_ventana INTEGER NOT NULL
        DEFAULT 1,

    verificado BOOLEAN NOT NULL
        DEFAULT FALSE,

    comprobante_hash VARCHAR(64),

    comprobante_expira_en TIMESTAMPTZ

);


-- ============================================================
-- POSTULACIONES DE ADOPCIÓN
-- ============================================================

CREATE TABLE IF NOT EXISTS public.postulaciones (

    id BIGSERIAL PRIMARY KEY,

    mascota_id INTEGER NOT NULL
        REFERENCES public.mascotas(id)
        ON DELETE CASCADE,

    adoptante_id INTEGER NOT NULL
        REFERENCES public.usuarios(id)
        ON DELETE CASCADE,

    telefono VARCHAR(25) NOT NULL,

    vivienda VARCHAR(20) NOT NULL
        CHECK (
            vivienda IN (
                'casa',
                'departamento',
                'otro'
            )
        ),

    motivo TEXT NOT NULL,

    estado VARCHAR(20) NOT NULL
        DEFAULT 'pendiente'
        CHECK (
            estado IN (
                'pendiente',
                'aprobada',
                'rechazada'
            )
        ),

    creada_en TIMESTAMPTZ NOT NULL
        DEFAULT NOW(),

    resuelta_en TIMESTAMPTZ,

    CONSTRAINT postulaciones_mascota_adoptante_unique
        UNIQUE (mascota_id, adoptante_id)

);


-- ============================================================
-- PREFERENCIAS DE MATCH
-- ============================================================

CREATE TABLE IF NOT EXISTS public.preferencias_match (

    usuario_id INTEGER PRIMARY KEY
        REFERENCES public.usuarios(id)
        ON DELETE CASCADE,

    tipo_preferido VARCHAR(20) NOT NULL
        CHECK (
            tipo_preferido IN (
                'perro',
                'gato',
                'otro',
                'indiferente'
            )
        ),

    personalidad_preferida VARCHAR(40) NOT NULL
        CHECK (
            personalidad_preferida IN (

                'tranquilo',
                'carinoso',
                'jugueton',
                'activo',
                'timido',
                'sociable',
                'independiente',
                'protector',

                'tranquilo_carinoso',
                'activo_jugueton',
                'jugueton_sociable',
                'timido_carinoso',
                'independiente_tranquilo',

                'indiferente'
            )
        ),

    nivel_cuidados VARCHAR(25) NOT NULL
        CHECK (
            nivel_cuidados IN (
                'basicos',
                'tratamientos',
                'especiales',
                'indiferente'
            )
        ),

    creada_en TIMESTAMPTZ NOT NULL
        DEFAULT NOW(),

    actualizada_en TIMESTAMPTZ NOT NULL
        DEFAULT NOW()

);


-- ============================================================
-- ÍNDICES
-- ============================================================

CREATE INDEX IF NOT EXISTS
idx_mascotas_usuario
ON public.mascotas (usuario_id);

CREATE INDEX IF NOT EXISTS
idx_mascotas_estado
ON public.mascotas (estado_adopcion);

CREATE INDEX IF NOT EXISTS
idx_mascotas_tipo
ON public.mascotas (tipo_mascota);

CREATE INDEX IF NOT EXISTS
idx_mascotas_personalidad
ON public.mascotas (personalidad_codigo);

CREATE INDEX IF NOT EXISTS
idx_mascotas_salud
ON public.mascotas (salud_codigo);

CREATE INDEX IF NOT EXISTS
idx_sesiones_usuario
ON public.sesiones (usuario_id);

CREATE INDEX IF NOT EXISTS
idx_sesiones_ultima_actividad
ON public.sesiones (ultima_actividad);

CREATE INDEX IF NOT EXISTS
idx_postulaciones_mascota
ON public.postulaciones (mascota_id);

CREATE INDEX IF NOT EXISTS
idx_postulaciones_adoptante
ON public.postulaciones (adoptante_id);

CREATE INDEX IF NOT EXISTS
idx_postulaciones_estado
ON public.postulaciones (estado);

CREATE INDEX IF NOT EXISTS
idx_preferencias_tipo
ON public.preferencias_match (tipo_preferido);

CREATE INDEX IF NOT EXISTS
idx_preferencias_personalidad
ON public.preferencias_match (personalidad_preferida);


COMMIT;


-- ============================================================
-- VERIFICACIÓN DE TABLAS
-- ============================================================

SELECT
    table_name
FROM information_schema.tables
WHERE table_schema = 'public'
  AND table_type = 'BASE TABLE'
ORDER BY table_name;
